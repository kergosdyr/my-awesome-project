    @Test void observationHarnessSeesUncommittedWriteAndRollback() throws Exception {
        seed(false);
        var hold=new Hold(true);
        var tx=new org.springframework.transaction.support.TransactionTemplate(
            second.getBean(org.springframework.transaction.PlatformTransactionManager.class));
        var otherJdbc=second.getBean(JdbcTemplate.class);
        var pending=workers.submit(()->tx.execute(status->{
            assertTrue(org.springframework.transaction.support.TransactionSynchronizationManager.isActualTransactionActive());
            // 인프라 검사용 단일 쓰기. 업무 메서드를 대신 구현하지 않는다.
            otherJdbc.update("UPDATE event SET remaining=0 WHERE event_id=1");
            hold.flushed("infrastructure");
            return 0;
        }));
        try {
            await(hold.reached); state(0,0,1);
            hold.release.countDown();
            var error=assertThrows(ExecutionException.class,()->pending.get(10,TimeUnit.SECONDS));
            assertInstanceOf(InjectedFailure.class,error.getCause()); state(0,0,1);
        } finally { hold.release.countDown(); }
    }
    private void seed(boolean full) {
        jdbc.update("INSERT INTO event(event_id,remaining) VALUES (1,?)",full?0:1);
        if(full) jdbc.update("INSERT INTO reservation(reservation_id,event_id,user_id) VALUES (10,1,100)");
    }
    private void state(int oldCount,int newCount,int remaining) {
        assertEquals(oldCount,jdbc.queryForObject("SELECT COUNT(*) FROM reservation WHERE user_id=100",Integer.class));
        assertEquals(newCount,jdbc.queryForObject("SELECT COUNT(*) FROM reservation WHERE user_id=200",Integer.class));
        assertEquals(remaining,jdbc.queryForObject("SELECT remaining FROM event WHERE event_id=1",Integer.class));
        assertEquals(1,remaining+jdbc.queryForObject("SELECT COUNT(*) FROM reservation",Integer.class));
    }

    static class InjectedFailure extends RuntimeException {}
    @Test void cancellationFailureAfterFlushRollsBack() {
        seed(true); var probe=second.getBean(TransactionProbe.class);
        probe.install(new TransactionProbe.Observer() {
            public void flushed(String operation) { throw new InjectedFailure(); }
        });
        try { assertThrows(InjectedFailure.class,()->serviceB.cancel(1,10)); }
        finally { probe.clear(); }
        state(1,0,0);
    }

    // 테스트 인프라: 별도 스레드에서 Spring 프록시를 호출해 별도 트랜잭션을 실행한다.
    private final ExecutorService workers=Executors.newFixedThreadPool(2);
    @AfterEach void stopWorkers() throws Exception {
        workers.shutdownNow(); assertTrue(workers.awaitTermination(5,TimeUnit.SECONDS));
    }
    private Future<MixedReservationService.Result> async(MixedReservationService service,
            TransactionProbe probe,TransactionProbe.Observer observer,
            Callable<MixedReservationService.Result> action) {
        return workers.submit(()->{
            probe.install(observer);
            try { return action.call(); } finally { probe.clear(); }
        });
    }
    private static void await(CountDownLatch latch) {
        try { assertTrue(latch.await(5,TimeUnit.SECONDS),"관찰 지점에 도달하지 못했습니다"); }
        catch(InterruptedException e) { Thread.currentThread().interrupt(); throw new AssertionError(e); }
    }
    private static MixedReservationService.Result result(Future<MixedReservationService.Result> future) throws Exception {
        return future.get(10,TimeUnit.SECONDS);
    }
    // flush 후 commit 직전에 멈춘다. flush를 commit으로 오해하지 않도록 별도 지점으로 제공.
    static class Hold implements TransactionProbe.Observer {
        final CountDownLatch reached=new CountDownLatch(1), release=new CountDownLatch(1);
        final boolean fail;
        Hold(boolean fail) { this.fail=fail; }
        public void flushed(String operation) {
            reached.countDown(); await(release);
            if(fail) throw new InjectedFailure();
        }
    }
    @Test @Tag("concurrency") void bothTransactionsEnterTogether() throws Exception {
        seed(true); var entered=new CountDownLatch(2);
        TransactionProbe.Observer barrier=new TransactionProbe.Observer() {
            public void entered(String operation) { entered.countDown(); await(entered); }
        };
        var c=async(serviceA,firstProbe(),barrier,()->serviceA.cancel(1,10));
        var r=async(serviceB,secondProbe(),barrier,()->serviceB.reserve(1,200));
        assertEquals(CANCELLED,result(c)); assertAllowedReservation(result(r));
    }
    @Test @Tag("concurrency") void cancellationFlushedButNotCommittedWhenReservationEnters() throws Exception {
        overlap(false);
    }
    @Test @Tag("concurrency") void cancellationRollsBackWhileReservationIsInFlight() throws Exception {
        overlap(true);
    }
    private void overlap(boolean fail) throws Exception {
        seed(true); var hold=new Hold(fail); var reserveEntered=new CountDownLatch(1);
        var c=async(serviceA,firstProbe(),hold,()->serviceA.cancel(1,10));
        try {
            if(!hold.reached.await(5,TimeUnit.SECONDS)) {
                result(c); // TODO/업무 예외가 먼저 났다면 타이밍 실패 대신 실제 원인을 표시한다.
                fail("flush 관찰 지점에 도달하지 않았습니다");
            }
            // 다른 연결에서는 아직 커밋되지 않은 취소가 보이지 않아야 한다.
            state(1,0,0);
            var r=async(serviceB,secondProbe(),new TransactionProbe.Observer() {
                public void entered(String operation) { reserveEntered.countDown(); }
            },()->serviceB.reserve(1,200));
            await(reserveEntered); hold.release.countDown();
            if(fail) {
                var error=assertThrows(ExecutionException.class,()->result(c));
                assertInstanceOf(InjectedFailure.class,error.getCause());
                assertEquals(SOLD_OUT,result(r)); state(1,0,0);
            } else {
                assertEquals(CANCELLED,result(c)); assertAllowedReservation(result(r));
            }
        } finally { hold.release.countDown(); }
    }
    private void assertAllowedReservation(MixedReservationService.Result value) {
        assertTrue(value==RESERVED || value==SOLD_OUT,"예외/NOT_FOUND는 허용 응답이 아닙니다");
        if(value==RESERVED) state(0,1,0); else state(0,0,1);
    }
    @Autowired private TransactionProbe probeA;
    private TransactionProbe firstProbe() { return probeA; }
    private TransactionProbe secondProbe() { return second.getBean(TransactionProbe.class); }
